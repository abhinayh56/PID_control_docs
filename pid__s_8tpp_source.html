44template <typename T>
45T control_system::pid::PID_S<T>::update(T x_0, T x, T u_ff_)
46{
47 T e_k = x_0 - x;
48 u_ff = u_ff_;
49
50 P = Kp * e_k;
51
52 I = I + (Kp / T_i) * e_k * dt;
53 I = saturate(I, -I_max, I_max);
54
55 if (start == true)
56 {
57 D = 0.0;
58 start = false;
59 }
60 else
61 {
62 D = Kp * T_d * (e_k - e_k_1) / dt;
63 }
64
65 if (d_filter == true)
66 {
67 lpf.cal_y(D);
68 D = lpf.get_y();
69 }
70 e_k_1 = e_k;
71 u = u_ff + P + I + D;
72 u = saturate(u, -u_max, u_max);
73 return u;
74}
